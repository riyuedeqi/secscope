cmake_minimum_required(VERSION 3.16)
project(secscope_cache LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(nlohmann_json REQUIRED)
add_library(cache_common protocol.cpp storage.cpp)
target_link_libraries(cache_common PUBLIC nlohmann_json::nlohmann_json)
add_executable(cache-daemon cache-daemon.cpp)
add_executable(cache-client cache-client.cpp)
target_link_libraries(cache-daemon PRIVATE cache_common)
target_link_libraries(cache-client PRIVATE cache_common)
install(TARGETS cache-daemon cache-client RUNTIME DESTINATION bin)
