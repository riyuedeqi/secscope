cmake_minimum_required(VERSION 3.16)
project(secscope_scanner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wconversion)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(scanner scanner.cpp file_inventory.cpp hashing.cpp resource_budget.cpp worker_pool.cpp process_snapshot.cpp socket_snapshot.cpp checkpoint.cpp native_io.cpp native_helpers.cpp runtime_snapshot.cpp file_monitor.cpp candidate_discovery.cpp filesystem_facts.cpp events_parser.cpp web_access_parser.cpp binary_analyzer.cpp static_analyzer.cpp process_control.cpp)
target_include_directories(scanner PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
find_package(ZLIB REQUIRED)
target_link_libraries(scanner PRIVATE OpenSSL::Crypto nlohmann_json::nlohmann_json ZLIB::ZLIB archive pthread)
install(TARGETS scanner RUNTIME DESTINATION bin)
